Step 3 — The Task Store

A simple in-memory store protected by a sync.RWMutex — safe for concurrent reads from multiple goroutines (remember lesson 11: each HTTP request runs in its own goroutine).

internal/store/tasks.go
package store

import (
    "fmt"
    "sync"
)

type Task struct {
    ID    string `json:"id"`
    Title string `json:"title"`
    Done  bool   `json:"done"`
}

type Store struct {
    mu     sync.RWMutex   // protects tasks
    tasks  map[string]Task
    nextID int
}

func New() *Store {
    return &Store{tasks: make(map[string]Task)}
}

func (s *Store) Add(title string) Task {
    s.mu.Lock()
    defer s.mu.Unlock()

    s.nextID++
    t := Task{ID: fmt.Sprintf("%d", s.nextID), Title: title}
    s.tasks[t.ID] = t
    return t
}

func (s *Store) Get(id string) (Task, bool) {
    s.mu.RLock()          // read-only lock: many readers allowed
    defer s.mu.RUnlock()
    t, ok := s.tasks[id]
    return t, ok
}

func (s *Store) List() []Task {
    s.mu.RLock()
    defer s.mu.RUnlock()
    out := make([]Task, 0, len(s.tasks))
    for _, t := range s.tasks {
        out = append(out, t)
    }
    return out
}

Step 4 — The Handlers

Handlers are methods on a Handler struct, so they share access to the store without globals.

internal/handlers/tasks.go
package handlers

import (
    "encoding/json"
    "net/http"

    "github.com/amit/tasks-api/internal/store"
)

type Handler struct {
    Store *store.Store
}

// writeJSON encodes v as JSON and sets Content-Type.
// Centralise this — don't repeat in every handler.
func writeJSON(w http.ResponseWriter, status int, v any) {
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(status)
    json.NewEncoder(w).Encode(v)
}

// writeError returns {"error": "message"} with the given status.
func writeError(w http.ResponseWriter, status int, msg string) {
    writeJSON(w, status, map[string]string{"error": msg})
}

// GET /tasks
func (h *Handler) ListTasks(w http.ResponseWriter, r *http.Request) {
    writeJSON(w, http.StatusOK, h.Store.List())
}

// POST /tasks — body: {"title": "..."}
func (h *Handler) CreateTask(w http.ResponseWriter, r *http.Request) {
    var body struct {
        Title string `json:"title"`
    }
    if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
        writeError(w, http.StatusBadRequest, "invalid JSON")
        return
    }
    if body.Title == "" {
        writeError(w, http.StatusUnprocessableEntity, "title is required")
        return
    }

    task := h.Store.Add(body.Title)
    writeJSON(w, http.StatusCreated, task)
}

// GET /tasks/{id}
func (h *Handler) GetTask(w http.ResponseWriter, r *http.Request) {
    id := r.PathValue("id")  // Go 1.22+ ServeMux path values
    task, ok := h.Store.Get(id)
    if !ok {
        writeError(w, http.StatusNotFound, "task not found")
        return
    }
    writeJSON(w, http.StatusOK, task)
}
r.PathValue("id") — added in Go 1.22. go.mod must declare go 1.22 or later. Reference →

Step 5 — Logging Middleware

internal/middleware/logging.go
package middleware

import (
    "log"
    "net/http"
    "time"
)

// Logging wraps an http.Handler and logs each request's
// method, path, and elapsed time.
func Logging(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()
        next.ServeHTTP(w, r)
        log.Printf("%s %s %s", r.Method, r.URL.Path, time.Since(start))
    })
}

Middleware signature: func(http.Handler) http.Handler. The inner function wraps the next handler — before/after logic is just code before and after next.ServeHTTP(w, r).

Step 6 — main.go: Wire It Together

cmd/server/main.go
package main

import (
    "context"
    "log"
    "net/http"
    "os"
    "os/signal"
    "syscall"
    "time"

    "github.com/amit/tasks-api/internal/handlers"
    "github.com/amit/tasks-api/internal/middleware"
    "github.com/amit/tasks-api/internal/store"
)

func main() {
    // 1. Build dependencies
    s   := store.New()
    h   := &handlers.Handler{Store: s}

    // 2. Register routes (Go 1.22 pattern: METHOD /path/{param})
    mux := http.NewServeMux()
    mux.HandleFunc("GET  /tasks",       h.ListTasks)
    mux.HandleFunc("POST /tasks",       h.CreateTask)
    mux.HandleFunc("GET  /tasks/{id}",  h.GetTask)

    // 3. Wrap with middleware
    handler := middleware.Logging(mux)

    // 4. Configure the server
    srv := &http.Server{
        Addr:         ":8080",
        Handler:      handler,
        ReadTimeout:  5 * time.Second,
        WriteTimeout: 10 * time.Second,
        IdleTimeout:  60 * time.Second,
    }

    // 5. Start in a goroutine so we can catch OS signals
    go func() {
        log.Println("listening on :8080")
        if err := srv.ListenAndServe(); err != http.ErrServerClosed {
            log.Fatalf("server error: %v", err)
        }
    }()

    // 6. Graceful shutdown on SIGINT / SIGTERM
    quit := make(chan os.Signal, 1)
    signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
    <-quit  // block until signal received

    ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
    defer cancel()

    log.Println("shutting down…")
    if err := srv.Shutdown(ctx); err != nil {
        log.Fatalf("shutdown error: %v", err)
    }
    log.Println("done")
}

Step 7 — One Handler Test

internal/handlers/tasks_test.go
package handlers_test

import (
    "encoding/json"
    "net/http"
    "net/http/httptest"
    "strings"
    "testing"

    "github.com/amit/tasks-api/internal/handlers"
    "github.com/amit/tasks-api/internal/store"
)

func TestCreateTask(t *testing.T) {
    h := &handlers.Handler{Store: store.New()}

    // --- 201 Created with valid body ---
    body := strings.NewReader(`{"title":"buy milk"}`)
    req  := httptest.NewRequest(http.MethodPost, "/tasks", body)
    req.Header.Set("Content-Type", "application/json")
    rec  := httptest.NewRecorder()

    h.CreateTask(rec, req)

    if rec.Code != http.StatusCreated {
        t.Fatalf("want 201, got %d", rec.Code)
    }

    var task struct{ Title string }
    json.NewDecoder(rec.Body).Decode(&task)
    if task.Title != "buy milk" {
        t.Errorf("want title %q, got %q", "buy milk", task.Title)
    }

    // --- 422 when title is empty ---
    req2 := httptest.NewRequest(http.MethodPost, "/tasks",
        strings.NewReader(`{"title":""}`))
    rec2 := httptest.NewRecorder()
    h.CreateTask(rec2, req2)

    if rec2.Code != http.StatusUnprocessableEntity {
        t.Fatalf("want 422, got %d", rec2.Code)
    }
}

httptest.NewRecorder() is Go's built-in test response writer. No real HTTP connections, no ports — fast and self-contained. net/http/httptest →
